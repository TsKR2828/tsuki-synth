def load_module(name, src):
    mod = types.ModuleType(name)
    mod.__file__ = str(ROOT / "tools" / "stem_verify.py")
    exec(compile(src, mod.__file__, "exec"), mod.__dict__)
    return mod


spec = importlib.util.spec_from_file_location("t_stem_stream", ROOT / "tests" / "test_stem_stream.py")

spec.loader.exec_module(tmod)
tests = [(n, f) for n, f in vars(tmod).items() if n.startswith("test_") and callable(f)]


class MP:
    """minimal monkeypatch stand-in"""
    def __init__(self):
        self._undo = []

    def setattr(self, obj, name, value):
        self._undo.append((obj, name, getattr(obj, name)))
        setattr(obj, name, value)

    def undo(self):
        for obj, name, old in reversed(self._undo):
            setattr(obj, name, old)


def run_all(sv_mod):
    tmod.sv = sv_mod
    results = {}
    for name, fn in tests:
        params = inspect.signature(fn).parameters
        variants = [{}]
        if "odd_size" in params:
            variants = [{"odd_size": k} for k in (1, 5, 7)]
        for v in variants:
            label = name + ("[%s]" % v["odd_size"] if v else "")
            mp = MP()
            with tempfile.TemporaryDirectory() as td:
                kwargs = dict(v)
                if "tmp_path" in params:
                    kwargs["tmp_path"] = Path(td)
                if "monkeypatch" in params:
                    kwargs["monkeypatch"] = mp
                try:
                    fn(**kwargs)
                    results[label] = "pass"
                except BaseException as e:  # noqa: BLE001
                    results[label] = "FAIL(%s)" % type(e).__name__
                finally:
                    mp.undo()
    return results
