"""WF0925-P1 scratch check for engineering-gaps:E17.
(1) find_cli() in tools/physics_verify.py and tools/verify_score.py on the
    real repo: which CLI is chosen, printed once per process.
(2) The same two find_cli() bodies exec'd with __file__ pointing into a
    throw-away fake repo, to show Release beats a NEWER Debug build and that
    the choice does not move when mtimes are flipped (old rule shown too).
(3) melody_roll_video.resolve_ffmpeg() lookup order.
Nothing in the repo is written."""
import contextlib
import importlib.util
import io
import os
import sys
import tempfile
import time
import types
from pathlib import Path

ROOT = Path(r"<repo>")
sys.path.insert(0, str(ROOT / "tools"))
fails = 0


def check(cond, msg):
    global fails
    print(("  PASS  " if cond else "  FAIL  ") + msg)
    if not cond:
        fails += 1


def load(name, path, fake_file=None):
    src = Path(path).read_text(encoding="utf-8")
    mod = types.ModuleType(name)
    mod.__file__ = str(fake_file or path)
    sys.modules[name] = mod
    exec(compile(src, str(path), "exec"), mod.__dict__)
    return mod


def old_rule(build):
    cands = list(build.rglob("TsukiSynthCLI.exe")) + list(build.rglob("tsukisynth-cli*")) \
        + list(build.rglob("TsukiSynthCLI"))
    cands = [c for c in cands if c.is_file()]
    return max(cands, key=lambda p: p.stat().st_mtime) if cands else None


print("[1] real repo")
for tool in ("physics_verify", "verify_score"):
    mod = load("chk_" + tool, ROOT / "tools" / (tool + ".py"))
    buf = io.StringIO()
    with contextlib.redirect_stdout(buf):
        first = mod.find_cli()
        second = mod.find_cli()
    out = buf.getvalue()
    print("  %s.find_cli() -> %s" % (tool, first))
    print("  stdout: %r" % out)
    check(first == second == ROOT / "build" / "TsukiSynthCLI_artefacts" / "Release" / "TsukiSynthCLI.exe",
          "%s picks build/TsukiSynthCLI_artefacts/Release/TsukiSynthCLI.exe" % tool)
    check(out.count("[find_cli] TsukiSynthCLI selected:") == 1 and str(first) in out,
          "%s prints the chosen path to stdout exactly once for two calls" % tool)

print("\n[2] fake repo: Debug newer than Release")
with tempfile.TemporaryDirectory(prefix="p1_findcli_") as td:
    fake = Path(td)
    (fake / "tools").mkdir()
    build = fake / "build"
    rel = build / "TsukiSynthCLI_artefacts" / "Release" / "TsukiSynthCLI.exe"
    dbg = build / "TsukiSynthCLI_artefacts" / "Debug" / "TsukiSynthCLI.exe"
    deep_rel = build / "sub" / "x" / "Release" / "TsukiSynthCLI.exe"
    for p in (rel, dbg, deep_rel):
        p.parent.mkdir(parents=True, exist_ok=True)
        p.write_bytes(b"stub")
    now = time.time()
    os.utime(rel, (now - 3600, now - 3600))
    os.utime(deep_rel, (now - 1800, now - 1800))
    os.utime(dbg, (now, now))
    for tool in ("physics_verify", "verify_score"):
        mod = load("fake_" + tool, ROOT / "tools" / (tool + ".py"), fake / "tools" / (tool + ".py"))
        with contextlib.redirect_stdout(io.StringIO()) as buf:
            chosen = mod.find_cli()
        print("  %s: old max(mtime) rule -> %s" % (tool, old_rule(build).relative_to(fake)))
        print("  %s: new find_cli()      -> %s   stdout=%r" % (tool, chosen.relative_to(fake), buf.getvalue().strip()))
        check(chosen == rel, "%s: Release (shallowest) chosen although Debug is newest" % tool)
        # flip the mtimes: the choice must not move
        os.utime(rel, (now + 60, now + 60))
        os.utime(dbg, (now - 7200, now - 7200))
        os.utime(deep_rel, (now + 120, now + 120))
        with contextlib.redirect_stdout(io.StringIO()):
            again = mod.find_cli()
        check(again == rel, "%s: same choice after flipping mtimes (not mtime-decided)" % tool)
        os.utime(rel, (now - 3600, now - 3600))
        os.utime(deep_rel, (now - 1800, now - 1800))
        os.utime(dbg, (now, now))
    # only a Debug build present -> still found (falls back, and says so)
    rel.unlink()
    deep_rel.unlink()
    mod = load("fake2_verify_score", ROOT / "tools" / "verify_score.py", fake / "tools" / "verify_score.py")
    with contextlib.redirect_stdout(io.StringIO()) as buf:
        only = mod.find_cli()
    check(only == dbg, "only Debug present -> Debug returned (printed: %r)" % buf.getvalue().strip())
    dbg.unlink()
    with contextlib.redirect_stdout(io.StringIO()) as buf:
        none = mod.find_cli()
    check(none is None and buf.getvalue() == "", "no candidate -> None, nothing printed")

print("\n[3] melody_roll_video.resolve_ffmpeg() order")
mrv = load("chk_mrv", ROOT / "tools" / "melody_roll_video.py")
saved_env = os.environ.pop("TSUKI_FFMPEG", None)
saved_which = mrv.shutil.which
try:
    r = mrv.resolve_ffmpeg("D:/x/ffmpeg.exe")
    check(r == ("D:/x/ffmpeg.exe", "--ffmpeg"), "--ffmpeg wins: %r" % (r,))
    os.environ["TSUKI_FFMPEG"] = "E:/env/ffmpeg.exe"
    r = mrv.resolve_ffmpeg("D:/x/ffmpeg.exe")
    check(r[1] == "--ffmpeg", "--ffmpeg still beats TSUKI_FFMPEG: %r" % (r,))
    r = mrv.resolve_ffmpeg(None)
    check(r == ("E:/env/ffmpeg.exe", "TSUKI_FFMPEG"), "TSUKI_FFMPEG beats PATH: %r" % (r,))
    del os.environ["TSUKI_FFMPEG"]
    r = mrv.resolve_ffmpeg(None)
    print("  this machine, no --ffmpeg/TSUKI_FFMPEG -> %r" % (r,))
    check(r[1] == "PATH" and Path(r[0]).resolve() == Path(str(mrv.DEFAULT_FFMPEG)).resolve(),
          "PATH hit here is the same file as the old hard-coded path (behaviour unchanged on this machine)")
    mrv.shutil.which = lambda _name: None
    r = mrv.resolve_ffmpeg(None)
    check(r == (str(mrv.DEFAULT_FFMPEG), "DEFAULT_FFMPEG"), "nothing else -> hard-coded fallback: %r" % (r,))
finally:
    mrv.shutil.which = saved_which
    if saved_env is not None:
        os.environ["TSUKI_FFMPEG"] = saved_env

print("\nRESULT: %s (%d failed checks)" % ("ALL PASS" if fails == 0 else "FAILURES", fails))
sys.exit(1 if fails else 0)
