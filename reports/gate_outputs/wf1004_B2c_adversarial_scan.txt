def load(f):
    d={}
    for l in open(f):
        t=l.rstrip('\n').split('\t'); kv=dict(x.split('=',1) for x in t[1:] if '=' in x)
        d[t[0]]=(float(kv['peak_dBFS']),float(kv['RMS_dBFS']))
    return d
