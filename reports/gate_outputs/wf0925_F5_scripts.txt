# -*- coding: utf-8 -*-
"""WF0925-F5: decompose the F5 residual energy of one probe render.

Re-implements NOTHING new for the metric itself: the residual is computed by
importing physics_verify.measure_residual_energy() (same window, same bands,
same Hann, same analysis limit). This script only adds *descriptive*
break-downs (where in frequency the residual power sits; how much a pure
theory synthesis from --dump-modes gives), none of which is a GATE.

Usage: python analyze_residual.py --tools <dir> --run <run dir> --eng piano
"""
import argparse, json, math, sys
from pathlib import Path
import numpy as np

ap = argparse.ArgumentParser()
ap.add_argument("--tools", required=True)
ap.add_argument("--run", required=True)
ap.add_argument("--eng", default="piano")
ap.add_argument("--top", type=int, default=12)
a = ap.parse_args()
sys.path.insert(0, str(Path(a.tools).resolve()))
import physics_verify as pv  # noqa: E402

run = Path(a.run)
stem = f"probe_{a.eng}_60_85_20_resid"
sr, x = pv.read_wav_mono(run / (stem + ".wav"))
dump = json.loads((run / (stem + ".dump.json")).read_text(encoding="utf-8"))
ev = dump["events"][0]
strings = ev.get("strings") or [ev["partials"]]
mode_freqs = sorted({float(m["freq"]) for v in strings for m in v})
noteoff = 2.0 * pv.T60_NOTEOFF_RATIO

rdb = pv.measure_residual_energy(sr, x, mode_freqs, noteoff)
print(f"[{run.name}/{a.eng}] physics_verify.measure_residual_energy = {rdb:+.2f} dB re total")


def spectrum(sig):
    i0 = int(pv.EXCITER_NOISE_SKIP_S * sr)
    i1 = min(len(sig), int(noteoff * sr))
    seg = np.asarray(sig[i0:i1], dtype=np.float64) * np.hanning(i1 - i0)
    nfft = 1 << int(math.ceil(math.log2(len(seg))))
    spec = np.abs(np.fft.rfft(seg, nfft)) ** 2
    binhz = sr / nfft
    limit = pv.analysis_frequency_limit(sr)
    kmax = min(len(spec) - 1, int(limit / binhz))
    power = spec[1:kmax + 1]
    freqs = np.arange(1, kmax + 1) * binhz
    inband = np.zeros(len(power), dtype=bool)
    for f in mode_freqs:
        if not (0.0 < f <= limit * (1.0 + pv.RESIDUAL_BAND_HALF_WIDTH)):
            continue
        inband |= ((freqs >= f * (1.0 - pv.RESIDUAL_BAND_HALF_WIDTH))
                   & (freqs <= f * (1.0 + pv.RESIDUAL_BAND_HALF_WIDTH)))
    return freqs, power, inband


freqs, power, inband = spectrum(x)
total = power.sum()
resid = power[~inband].sum()
print(f"  re-computed here: {10*math.log10(resid/total):+.2f} dB (must equal above)")

# Region break-down: contiguous out-of-band runs.
out = ~inband
edges = np.flatnonzero(np.diff(np.concatenate([[0], out.astype(int), [0]])))
runs = list(zip(edges[0::2], edges[1::2]))
rows = []
for s, e in runs:
    p = power[s:e].sum()
    rows.append((p, freqs[s], freqs[e - 1]))
rows.sort(reverse=True)
print(f"  out-of-band regions: {len(rows)}; top {a.top} by power "
      f"(share of residual, dB re total):")
for p, f0, f1 in rows[:a.top]:
    print(f"    {f0:9.1f} - {f1:9.1f} Hz : {100*p/resid:6.2f}% of residual, "
          f"{10*math.log10(p/total):+7.2f} dB re total")

# Octave-ish coarse buckets of the residual.
buckets = [(0, 250), (250, 1000), (1000, 2000), (2000, 4000), (4000, 8000),
           (8000, 12000), (12000, 16000), (16000, 20001)]
print("  residual by coarse band (dB re total):")
for lo, hi in buckets:
    m = out & (freqs >= lo) & (freqs < hi)
    if not m.any():
        print(f"    {lo:6d}-{hi:6d} Hz: (no out-of-band bins: the +/-3% bands "
              f"of neighbouring partials overlap and cover this whole range)")
        continue
    p = power[m].sum()
    print(f"    {lo:6d}-{hi:6d} Hz: {10*math.log10(max(p,1e-300)/total):+8.2f} dB "
          f"({100*p/resid:6.2f}% of residual)")

# Pure theory synthesis from the dump (no exciter noise, no -60 dB stop).
dur = len(x) / sr
th = pv.synth_theory_signal(strings, sr, dur)
tf, tp, tin = spectrum(th)
tres = tp[~tin].sum() / tp.sum()
print(f"  theory synthesis (dump modes only) residual: {10*math.log10(tres):+.2f} dB re total")
print(f"  highest predicted mode: {max(mode_freqs):.1f} Hz; n modes {len(mode_freqs)}")

# -*- coding: utf-8 -*-
"""WF0925-F5: single-variable experiments at the THEORY level.

Premise (checked by analyze_residual.py): a pure synthesis of the --dump-modes
partials (physics_verify.synth_theory_signal(), no exciter noise, no -60 dB
mode stop) gives the SAME F5 residual as the real CLI render, to 0.01 dB,
both before and after the candidate. So the F5 number is a property of the
dumped (freq, amp, decay) set + the F5 metrology, and we can swap ONE of
those three quantities at a time between the before/after dumps.

The metric itself is physics_verify.measure_residual_energy() (imported,
unchanged). Everything printed here is descriptive, not a GATE.
"""
import argparse, copy, json, math, sys
from pathlib import Path
import numpy as np

ap = argparse.ArgumentParser()
ap.add_argument("--tools", required=True)
ap.add_argument("--before", required=True)
ap.add_argument("--after", required=True)
ap.add_argument("--eng", default="piano")
a = ap.parse_args()
sys.path.insert(0, str(Path(a.tools).resolve()))
import physics_verify as pv  # noqa: E402

SR, DUR = 48000, 2.2
NOTEOFF = 2.0 * pv.T60_NOTEOFF_RATIO
stem = f"probe_{a.eng}_60_85_20_resid.dump.json"


def load(run):
    ev = json.loads((Path(run) / stem).read_text(encoding="utf-8"))["events"][0]
    return ev["strings"]


def resid(strings):
    freqs = sorted({float(m["freq"]) for v in strings for m in v})
    x = pv.synth_theory_signal(strings, SR, DUR)
    return pv.measure_residual_energy(SR, x, freqs, NOTEOFF)


B, A = load(a.before), load(a.after)
assert len(B) == len(A) and all(len(b) == len(c) for b, c in zip(B, A))


def hybrid(base, donor, fields):
    out = copy.deepcopy(base)
    for s in range(len(out)):
        for i in range(len(out[s])):
            for f in fields:
                out[s][i][f] = donor[s][i][f]
    return out


print(f"== {a.eng}, MIDI 60 probe, theory synthesis of dumped modes ==")
print(f"before (all before)            : {resid(B):+7.2f} dB")
print(f"after  (all after)             : {resid(A):+7.2f} dB")
print("-- start from BEFORE, import ONE quantity from AFTER --")
for f in ("freq", "amp", "decay"):
    print(f"before + after {f:5s}           : {resid(hybrid(B, A, [f])):+7.2f} dB")
print("-- start from AFTER, restore ONE quantity from BEFORE --")
for f in ("freq", "amp", "decay"):
    print(f"after + before {f:5s}           : {resid(hybrid(A, B, [f])):+7.2f} dB")

# Only the fundamental's decay (all 3 strings) swapped.
def swap_decay_index(base, donor, idxs):
    out = copy.deepcopy(base)
    for s in range(len(out)):
        for i in idxs:
            out[s][i]["decay"] = donor[s][i]["decay"]
    return out

print("-- decay swaps restricted to specific partial indices --")
print(f"before + after decay of n=1 only       : {resid(swap_decay_index(B, A, [0])):+7.2f} dB")
print(f"before + after decay of n=1,2          : {resid(swap_decay_index(B, A, [0, 1])):+7.2f} dB")
print(f"after + before decay of n=1 only       : {resid(swap_decay_index(A, B, [0])):+7.2f} dB")
print(f"after + before decay of n=1,2          : {resid(swap_decay_index(A, B, [0, 1])):+7.2f} dB")

# Per-partial-index contribution to the residual (all strings of index n),
# relative to the FULL signal's total energy in the same window.
def spectrum(x, freqs):
    i0 = int(pv.EXCITER_NOISE_SKIP_S * SR); i1 = int(NOTEOFF * SR)
    seg = x[i0:i1] * np.hanning(i1 - i0)
    nfft = 1 << int(math.ceil(math.log2(len(seg))))
    spec = np.abs(np.fft.rfft(seg, nfft)) ** 2
    binhz = SR / nfft
    lim = pv.analysis_frequency_limit(SR)
    kmax = min(len(spec) - 1, int(lim / binhz))
    p = spec[1:kmax + 1]; fr = np.arange(1, kmax + 1) * binhz
    inb = np.zeros(len(p), dtype=bool)
    for f in freqs:
        if 0 < f <= lim * (1 + pv.RESIDUAL_BAND_HALF_WIDTH):
            inb |= (fr >= f * (1 - pv.RESIDUAL_BAND_HALF_WIDTH)) & (fr <= f * (1 + pv.RESIDUAL_BAND_HALF_WIDTH))
    return p, inb

for tag, S in (("before", B), ("after", A)):
    freqs = sorted({float(m["freq"]) for v in S for m in v})
    ptot, inb = spectrum(pv.synth_theory_signal(S, SR, DUR), freqs)
    tot = ptot.sum()
    print(f"-- {tag}: residual contribution of each partial index alone "
          f"(dB re FULL-signal total; top 5) --")
    rows = []
    for i in range(len(S[0])):
        sub = [[S[s][i]] for s in range(len(S))]
        p, _ = spectrum(pv.synth_theory_signal(sub, SR, DUR), freqs)
        r = p[~inb].sum()
        rows.append((r, i + 1, S[1][i]["freq"], S[1][i]["amp"], S[1][i]["decay"],
                     p.sum()))
    rows.sort(reverse=True)
    for r, n, f, amp, dec, e in rows[:5]:
        print(f"    n={n:2d} f={f:8.2f} Hz amp={amp:.5f} T60={dec:.4f} s : "
              f"resid {10*math.log10(r/tot):+7.2f} dB re total; "
              f"own leak fraction {10*math.log10(r/e):+7.2f} dB re own energy; "
              f"own energy share {100*e/tot:5.1f}%")

# -*- coding: utf-8 -*-
"""WF0925-F5: split the MIDI 60 piano fundamental 1/T60 into the four loss
channels of StringModel::decayTimeForFrequency() (internal friction eta,
Cuesta&Valette air+visc+disl, bridge loss), for before (L=0.35*2^(-9/12),
d=0.8 mm score default) and after (table L=0.639 m, d=1.00 mm).

Pure Python mirror of the C++ formulas (tree/src/physics/StringModel.h,
MaterialDB.h) -- analysis only, cross-checked below against the --dump-modes
"decay" field of the actual CLI runs. Not a GATE.
"""
import json, math
from pathlib import Path

HERE = Path(__file__).resolve().parent
mats = json.loads((HERE / "tree/data/materials.json").read_text(encoding="utf-8"))["materials"]
steel, spruce = mats["steel"], mats["wood_spruce"]

AIR_RHO, AIR_NU, DISL = 1.225, 1.619e-5, 1.0 / 18000.0
ETA_TO_RATE, LN1000 = 2.2, 6.907755278982137
H = 0.009  # kBridgeSoundboardThicknessM


def q_air_visc_disl(f, r, T, m):
    w = 2 * math.pi * f
    M = (r * 0.5) * math.sqrt(w / AIR_NU)
    qa = (AIR_RHO / m["density"]) * (math.sqrt(2) / M + 1 / (2 * M * M))
    qv = 0.003 * m["youngs_modulus"] * m["density"] * math.pi ** 2 * r ** 6 * w * w / (4 * T * T)
    return qa, qv, DISL


def bridge_loss(T, L, sb):
    E, nu, rho = sb["youngs_modulus"], sb["poisson_ratio"], sb["density"]
    D = E * H ** 3 / (12 * (1 - nu * nu))
    rhoS = rho * H
    G = 1 / (8 * math.sqrt(D * rhoS))
    return T * G / (LN1000 * L)


def tension(midi, L, d, rho):
    f = 440 * 2 ** ((midi - 69) / 12)
    mu = rho * math.pi * (d / 2) ** 2
    return mu * (2 * L * f) ** 2


def B_coeff(L, d, T, E):
    return math.pi ** 3 * E * d ** 4 / (64 * T * L * L)


def report(tag, L, d, f=261.626):
    T = tension(60, L, d, steel["density"])
    r = d / 2
    qa, qv, qd = q_air_visc_disl(f, r, T, steel)
    eta = steel["damping"]["eta"]
    rates = {
        "eta (internal friction)": eta * f / ETA_TO_RATE,
        "Q_air": qa * f / ETA_TO_RATE,
        "Q_visc": qv * f / ETA_TO_RATE,
        "Q_disl": qd * f / ETA_TO_RATE,
        "bridgeLoss": bridge_loss(T, L, spruce),
    }
    tot = sum(rates.values())

# -*- coding: utf-8 -*-
"""WF0925-F5: how the F5 metric responds to decay time alone.

(1) One decaying sinusoid at 261.626 Hz, bands = its own +/-3%: the fraction
    of its windowed energy that F5 counts as 'residual' (pure window/decay
    leakage; the signal contains nothing but the predicted mode).
(2) The real MIDI 60 piano probe (before dump), with ONLY the bridge-loss
    channel scaled by k (per-mode T60 recomputed with the StringModel
    formula, mirrored in t60_terms.py), to show the F5 value vs the
    fundamental's T60 and where the existing -60 dB line is crossed.
All numbers are descriptive (not a GATE, no new threshold). The metric is
physics_verify.measure_residual_energy(), imported unchanged.
"""
import copy, json, math, sys
from pathlib import Path
import numpy as np

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE / "tools_before"))
import physics_verify as pv  # noqa: E402
sys.path.insert(0, str(HERE))
import t60_terms as tt  # noqa: E402  (prints its own report on import)

SR, DUR = 48000, 2.2
NOTEOFF = 2.0 * pv.T60_NOTEOFF_RATIO
F = 261.626

print("\n(1) single decaying sinusoid, f=261.626 Hz, bands = own +/-3%:")
print("    T60 (s) | F5 'residual' = own leakage (dB re own energy)")
for t60 in (1.0, 1.5, 2.0, 2.5, 2.6733, 2.8737, 3.0, 3.5, 3.8744, 4.1425,
            5.0, 6.0, 8.0, 10.0, 15.0, 20.0, 50.0):
    x = pv.synth_theory_signal([[{"freq": F, "amp": 1.0, "decay": t60}]], SR, DUR)
    r = pv.measure_residual_energy(SR, x, [F], NOTEOFF)
    print(f"    {t60:7.4f} | {r:+8.2f}")

# (2) bridge-loss scan on the real before dump.
ev = json.loads((HERE / "run_before_own/probe_piano_60_85_20_resid.dump.json")
                .read_text(encoding="utf-8"))["events"][0]
S0 = ev["strings"]
Lb = 0.35 * 2 ** (-(60 - 69) / 12)
db = 0.0008
Tb = tt.tension(60, Lb, db, tt.steel["density"])
bl0 = tt.bridge_loss(Tb, Lb, tt.spruce)
eta = tt.steel["damping"]["eta"]


def t60_of(f, k):
    qa, qv, qd = tt.q_air_visc_disl(f, db / 2, Tb, tt.steel)
    return 1.0 / ((eta + qa + qv + qd) * f / tt.ETA_TO_RATE + k * bl0)


# validate the mirror against the dump at k=1
maxrel = max(abs(t60_of(m["freq"], 1.0) - m["decay"]) / m["decay"]
             for v in S0 for m in v)
print(f"\n(2) mirror check: max |T60_python - T60_dump| / T60_dump over 120 modes "
      f"= {maxrel:.2e}")
freqs = sorted({float(m["freq"]) for v in S0 for m in v})
print("    bridgeLoss x k | fundamental T60 (s) | F5 piano residual (dB re total)")
rows = []
for k in (0.0, 0.25, 0.5, 0.75, 1.0, 1.1, 1.2, 1.3, 1.4, 1.5, 1.6, 1.6962, 1.8, 2.0, 2.5, 3.0):
    S = copy.deepcopy(S0)
    for v in S:
        for m in v:
            m["decay"] = t60_of(m["freq"], k)
    x = pv.synth_theory_signal(S, SR, DUR)
    r = pv.measure_residual_energy(SR, x, freqs, NOTEOFF)
    rows.append((k, S[1][0]["decay"], r))
    print(f"    {k:6.4f}        | {S[1][0]['decay']:8.4f}            | {r:+8.2f}")
# linear interpolation of the -60 dB crossing (descriptive only)
for (k1, t1, r1), (k2, t2, r2) in zip(rows, rows[1:]):
    if (r1 - (-60.0)) * (r2 - (-60.0)) <= 0 and r1 != r2:
        w = (-60.0 - r1) / (r2 - r1)
        print(f"    existing -60 dB line crossed between k={k1} and k={k2}: "
              f"fundamental T60 ~= {t1 + w*(t2-t1):.3f} s "
              f"(linear interpolation, descriptive, not a threshold)")

"""WF0925-F5: inharmonicity B (closed form B=(r-1)/(4-r), r=(f2/2f1)^2, the
same definition as reports/d11_string_scale_before_after.md §2) and
fundamental T60 for: before (d=1.0 mm score), full candidate (table L+d),
and the L-only variant (table L, score d=1.0 mm; variant CLI with
TSUKI_F5_EXP=L_only). Piano engine, steel, from --dump-modes string 0.
Descriptive only; A14 §3.10 measured B=2e-4 (G5/G6) quoted from the D11 report."""
import json, os, subprocess, sys
from pathlib import Path

sys.path.insert(0, str(HERE / "tools_before"))
import physics_verify as pv

       ("candidate", HERE / "cli_after/TsukiSynthCLI.exe", None),
       ("L-only d=1.0", HERE / "cli_variant/TsukiSynthCLI.exe", "L_only")]
def dump(cli, exp, midi):
    env = dict(os.environ)
    env.pop("TSUKI_F5_EXP", None)
    if exp: env["TSUKI_F5_EXP"] = exp
    pv.ENGINES["piano"]["params"] = {"material": "steel", "diameter_mm": 1.0}
    sf, _ = pv.build_probe_score("piano", midi, out, vel=0.45, dur=0.05, tag="_b")
    r = subprocess.run([str(cli), "--dump-modes", str(sf)], capture_output=True, text=True, env=env, timeout=600)
    p = json.loads(r.stdout)["events"][0]["partials"]
    f1, f2 = p[0]["freq"], p[1]["freq"]
    rr = (f2 / (2 * f1)) ** 2
    return (rr - 1) / (4 - rr), p[0]["decay"]
