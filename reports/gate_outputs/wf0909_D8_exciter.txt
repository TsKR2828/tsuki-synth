#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""WF0909-D8 Rule 10 before/after metrics for the two moonlight scores whose
tongue_drum events changed exciter finger -> wood_mallet.

Whole-piece metrics (identical definitions to reports/gate_outputs/b4_method/
render_affected_pieces.py, extended with a 3-band energy split):
  * Channels mixed to mono by averaging.
  * RMS dBFS = 20*log10(sqrt(mean(x^2))) over the ENTIRE file.
  * Spectral centroid = amplitude-weighted mean frequency of one rfft of the
    ENTIRE mono signal (no window).
  * Band energy % = sum(|X(f)|^2) in the band / sum(|X(f)|^2) over the whole
    spectrum * 100, bands <200 Hz / 200 Hz-2 kHz / >2 kHz (same edges as
    D8 decision packet SS3.5).

Usage: python rule10_metrics.py
"""
import wave
import numpy as np

FILES = {
    "tongue_drum": {
        "before": r"output\wf0909\D8\render_before\moonlight_sonata_i_tongue_drum.wav",
        "after": r"output\wf0909\D8\render_after\moonlight_sonata_i_tongue_drum.wav",
    },
    "yangqin_tongue_mix": {
        "before": r"output\wf0909\D8\render_before\moonlight_sonata_i_yangqin_tongue_mix.wav",
        "after": r"output\wf0909\D8\render_after\moonlight_sonata_i_yangqin_tongue_mix.wav",
    },
}


def read_wav_mono(path):
    with wave.open(path, "rb") as w:
        sr = w.getframerate()
        n = w.getnframes()
        sw = w.getsampwidth()
        ch = w.getnchannels()
        raw = w.readframes(n)
    if sw == 2:
        data = np.frombuffer(raw, dtype="<i2").astype(np.float64) / 32768.0
    elif sw == 3:
        b = np.frombuffer(raw, dtype=np.uint8).reshape(-1, 3)
        as_int = (b[:, 0].astype(np.int32)
                  | (b[:, 1].astype(np.int32) << 8)
                  | (b[:, 2].astype(np.int32) << 16))
        as_int[as_int >= (1 << 23)] -= (1 << 24)
        data = as_int.astype(np.float64) / (1 << 23)
    elif sw == 4:
        data = np.frombuffer(raw, dtype="<i4").astype(np.float64) / (2 ** 31)
    else:
        raise ValueError("unsupported sampwidth %d" % sw)
    if ch > 1:
        data = data.reshape(-1, ch).mean(axis=1)
    return sr, data


def rms_dbfs(x):
    r = np.sqrt(np.mean(x ** 2)) if len(x) else 0.0
    return 20 * np.log10(max(r, 1e-12))


def peak_dbfs(x):
    p = np.max(np.abs(x)) if len(x) else 0.0
    return 20 * np.log10(max(p, 1e-12))


def spectral_metrics(sr, x):
    n = len(x)
    X = np.abs(np.fft.rfft(x)) ** 2
    freqs = np.fft.rfftfreq(n, 1.0 / sr)
    total = X.sum()
    centroid = float((freqs * X).sum() / total) if total > 0 else float("nan")
    band_lt200 = 100.0 * X[freqs < 200].sum() / total
    band_mid = 100.0 * X[(freqs >= 200) & (freqs <= 2000)].sum() / total
    band_hi = 100.0 * X[freqs > 2000].sum() / total
    return centroid, band_lt200, band_mid, band_hi


def main():
    for piece, paths in FILES.items():
        print("=" * 78)
        print(piece)
        print("=" * 78)
        rows = {}
        for label, path in paths.items():
            sr, x = read_wav_mono(path)
            rms = rms_dbfs(x)
            peak = peak_dbfs(x)
            centroid, lt200, mid, hi = spectral_metrics(sr, x)
            rows[label] = dict(rms=rms, peak=peak, centroid=centroid,
                                lt200=lt200, mid=mid, hi=hi, len_s=len(x) / sr)
            print("%-8s len=%.3fs sr=%d  RMS=%.2f dBFS  peak=%.2f dBFS  "
                  "centroid=%.2f Hz  <200Hz=%.4f%%  200Hz-2kHz=%.4f%%  >2kHz=%.4f%%"
                  % (label, len(x) / sr, sr, rms, peak, centroid, lt200, mid, hi))
        d = rows["after"]
        b = rows["before"]
        print("  delta: dRMS=%.3f dB  dCentroid=%+.2f%%  <200Hz %.4f%%->%.4f%%  "
              "200Hz-2kHz %.4f%%->%.4f%%  >2kHz %.4f%%->%.4f%%"
              % (d["rms"] - b["rms"],
                 100.0 * (d["centroid"] - b["centroid"]) / b["centroid"],
                 b["lt200"], d["lt200"], b["mid"], d["mid"], b["hi"], d["hi"]))
        print()


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""WF0909-D8: measure MIDI 37->87 pitch-loudness slope for the 4 probe
renders. Methodology reused from output/wf0907/R6/verify3/measure.py
(D8 decision-packet round 3/4 method): notes at 8.0s spacing, RMS over a
0.3s window and a 6.0s window from each note's onset, slope = RMS(37) -
RMS(87). fund_share_pct_6s = % of the 6.0s window's spectral energy within
+-3% of the note's equal-tempered f0 (same tolerance as verify3/measure.py).
"""
import glob
import json
import os
import wave

import numpy as np

RD = r"output\wf0909\D8\probe_render"
MIDIS = [37, 47, 57, 67, 77, 87]
SPACING = 8.0


def rd(p):
    with wave.open(p, "rb") as w:
        n, ch, sw, sr = w.getnframes(), w.getnchannels(), w.getsampwidth(), w.getframerate()
        raw = w.readframes(n)
    if sw == 3:
        a = np.frombuffer(raw, dtype=np.uint8).reshape(-1, 3).astype(np.int32)
        v = (a[:, 0] | (a[:, 1] << 8) | (a[:, 2] << 16))
        v = np.where(v & 0x800000, v - 0x1000000, v)
        x = v.astype(np.float64) / 8388608.0
    elif sw == 2:
        x = np.frombuffer(raw, dtype="<i2").astype(np.float64) / 32768.0
    else:
        raise SystemExit("sw " + str(sw))
    x = x.reshape(-1, ch)
    return x.mean(axis=1), sr


def db(v):
    return 20 * np.log10(v) if v > 0 else float("-inf")


def fundshare(seg, sr, f0):
    n = len(seg)
    X = np.abs(np.fft.rfft(seg)) ** 2
    fr = np.fft.rfftfreq(n, 1 / sr)
    tot = X.sum()
    if tot <= 0:
        return float("nan")
    band = (fr >= f0 * 0.97) & (fr <= f0 * 1.03)
    return 100.0 * X[band].sum() / tot


def main():
    res = {}
    for p in sorted(glob.glob(os.path.join(RD, "*.wav"))):
        x, sr = rd(p)
        name = os.path.basename(p)[:-4]
        row = {}
        for i, m in enumerate(MIDIS):
            f0 = 440.0 * 2 ** ((m - 69) / 12.0)
            s = int(i * SPACING * sr)
            seg03 = x[s:s + int(0.3 * sr)]
            seg60 = x[s:s + int(6.0 * sr)]
            row[m] = {
                "rms_0p3": db(float(np.sqrt((seg03 ** 2).mean()))),
                "rms_6p0": db(float(np.sqrt((seg60 ** 2).mean()))),
                "peak": db(float(np.abs(seg60).max())),
                "fund_share_pct_6s": fundshare(seg60, sr, f0),
            }
        for k in ("rms_0p3", "rms_6p0", "peak"):
            row["slope_" + k] = row[MIDIS[0]][k] - row[MIDIS[-1]][k]
        res[name] = row
        print("%-24s 0.3s slope %7.2f dB | 6.0s slope %7.2f dB | fund%% 37=%6.2f 87=%6.2f"
              % (name, row["slope_rms_0p3"], row["slope_rms_6p0"],
                 row[37]["fund_share_pct_6s"], row[87]["fund_share_pct_6s"]))
    out_path = r"output\wf0909\D8\measure_slope.json"
    with open(out_path, "w", encoding="utf-8") as f:
        json.dump(res, f, indent=1)
    print("\nwrote", out_path)
    print()
    for name in sorted(res):
        print(name, " ".join("%.2f" % res[name][m]["rms_0p3"] for m in MIDIS),
              "| 6s:", " ".join("%.2f" % res[name][m]["rms_6p0"] for m in MIDIS))


if __name__ == "__main__":
    main()
