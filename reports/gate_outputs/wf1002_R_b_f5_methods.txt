"""WF1002-R R-b(2): F5 measurement-method study (READ-ONLY research; nothing in tools/ is changed).

Imports tools/physics_verify.py as a library and calls its OWN probe builder, --dump-modes reader,
theory synthesiser and measure_residual_energy() (= M0, the current F5).  Candidate methods are
implemented HERE only, for description.  All numbers are descriptive, NOT GATE, NOT new thresholds.

Usage (from output/wf1002/R): python scripts/rb_f5_methods.py bin/TsukiSynthCLI.exe f5_methods
"""

def band_mask(freqs, mode_freqs, limit):
    m = np.zeros(len(freqs), dtype=bool)
    for f in mode_freqs:
        if 0.0 < f <= limit * (1.0 + HW):
            m |= (freqs >= f * (1.0 - HW)) & (freqs <= f * (1.0 + HW))
    return m


def spec_power(seg, win):
    seg = seg * win
    nfft = 1 << int(math.ceil(math.log2(len(seg))))
    p = np.abs(np.fft.rfft(seg, nfft)) ** 2
    binhz = SR / nfft
    limit = pv.analysis_frequency_limit(SR)
    kmax = min(len(p) - 1, int(limit / binhz))
    return p[1:kmax + 1], np.arange(1, kmax + 1) * binhz, limit


def seg_of(x):
    i1 = min(len(x), int(NOTEOFF * SR)); return np.asarray(x[I0:i1], float)


def resid_db_window(x, mode_freqs, winname):
    seg = seg_of(x); n = len(seg)
    win = {"blackmanharris": windows.blackmanharris(n),
           "kaiser20": windows.kaiser(n, 20.0),
           "dpss4": windows.dpss(n, 4.0)}[winname]
    p, f, lim = spec_power(seg, win)
    m = band_mask(f, mode_freqs, lim)
    return 10 * math.log10(max(p[~m].sum(), 1e-300) / p.sum())


def resid_db_dedecay(x, mode_freqs, t60_ref):
    """M3: undo the predicted decay of the reference mode (string 0, mode 0) before Hann+FFT."""
    seg = seg_of(x)
    t = (np.arange(len(seg)) + I0) / SR
    seg = seg * np.exp(pv.MODAL_DECAY_LN1000 * t / t60_ref)
    p, f, lim = spec_power(seg, np.hanning(len(seg)))
    m = band_mask(f, mode_freqs, lim)
    return 10 * math.log10(max(p[~m].sum(), 1e-300) / p.sum())


def td_residual(x, model, mode_freqs):
    """M1: energy of (x - model) re energy of x, F5's Hann window / band limit, NO band exclusion.
    Also returns the share of that residual that lies inside the predicted +/-3% bands."""
    seg = seg_of(x); mod = seg_of(model)
    win = np.hanning(len(seg))
    p_tot, f, lim = spec_power(seg, win)
    p_res, _, _ = spec_power(seg - mod, win)
    m = band_mask(f, mode_freqs, lim)
    return 10 * math.log10(max(p_res.sum(), 1e-300) / p_tot.sum()), p_res[m].sum() / max(p_res.sum(), 1e-300)


def ls_fit_model(x, strings):
    """M1b: least-squares complex amplitude (sin+cos) per predicted mode, using ONLY the predicted
    frequency and T60 from --dump-modes (amplitudes/phases fitted on the segment)."""
    i1 = min(len(x), int(NOTEOFF * SR))
    t = np.arange(len(x)) / SR
    cols = []
    lim = pv.analysis_frequency_limit(SR)
    for voice in strings:
        for m in voice:
            if not (0 < m["freq"] <= lim):
                continue
            env = np.exp(-pv.MODAL_DECAY_LN1000 * t / max(m["decay"], 1e-9))
            cols.append(env * np.sin(2 * np.pi * m["freq"] * t))
            cols.append(env * np.cos(2 * np.pi * m["freq"] * t))
    A = np.stack(cols, axis=1)
    coef, *_ = np.linalg.lstsq(A[I0:i1], x[I0:i1], rcond=None)
    return A @ coef


def excess_db(x, theory_scaled, mode_freqs):
    """M8: out-of-band power of x MINUS out-of-band power of the scaled theory synthesis
    (the model's own leakage), re total power of x (Hann, same bands)."""
    seg = seg_of(x); th = seg_of(theory_scaled)
    win = np.hanning(len(seg))
    p, f, lim = spec_power(seg, win); pt, _, _ = spec_power(th, win)
    m = band_mask(f, mode_freqs, lim)
    ex = p[~m].sum() - pt[~m].sum()
    return (10 * math.log10(ex / p.sum()) if ex > 0 else float("-inf"))


def probe(eng, override, tag):
    wav, sf = pv.render_probe(CLI, eng, 60, OUT, params_override=override, tag=tag)
    sr, x = pv.read_wav_mono(wav)
    assert sr == SR
    ev = pv.dump_modes_event(CLI, sf)
    strings = ev.get("strings") or [ev["partials"]]
    mode_freqs = sorted({float(m["freq"]) for v in strings for m in v})
    theory = pv.synth_theory_signal(strings, SR, len(x) / SR)[:len(x)]
    i1 = int(NOTEOFF * SR)
    g = float(np.dot(x[I0:i1], theory[I0:i1]) / np.dot(theory[I0:i1], theory[I0:i1]))
    raw = subprocess.run([str(CLI), "--dump-modes", str(sf)], capture_output=True, text=True).stdout
    return x, strings, mode_freqs, theory, g, wav, raw


def all_methods(x, strings, mode_freqs, theory, g):
    t60_ref = strings[0][0]["decay"]
    r = {}
    r["M0_current_hann"] = pv.measure_residual_energy(SR, x, mode_freqs, NOTEOFF)
    r["M2a_blackmanharris"] = resid_db_window(x, mode_freqs, "blackmanharris")
    r["M2b_kaiser20"] = resid_db_window(x, mode_freqs, "kaiser20")
    r["M2c_dpss_nw4"] = resid_db_window(x, mode_freqs, "dpss4")
    r["M3_dedecay_f1"] = resid_db_dedecay(x, mode_freqs, t60_ref)
    r["M1a_minus_theory"], inb_a = td_residual(x, g * theory, mode_freqs)
    r["M1b_minus_LSfit"], inb_b = td_residual(x, ls_fit_model(x, strings), mode_freqs)
    r["M8_excess_over_theory_leak"] = excess_db(x, g * theory, mode_freqs)
    return r, (inb_a, inb_b)

def inj_signal(n, kind, f1):
    t = np.arange(n) / SR
    if kind == "stationary_3000Hz":
        return np.sin(2 * np.pi * 3000.0 * t)
    if kind == "decaying_1.5f1_outband":
        return np.exp(-pv.MODAL_DECAY_LN1000 * t / 2.87) * np.sin(2 * np.pi * 1.5 * f1 * t)
    if kind == "stationary_inband_1.015f1":
        return np.sin(2 * np.pi * 1.015 * f1 * t)
    raise ValueError(kind)


def hann_power(sig):
    seg = seg_of(sig); p, _, _ = spec_power(seg, np.hanning(len(seg))); return p.sum()
