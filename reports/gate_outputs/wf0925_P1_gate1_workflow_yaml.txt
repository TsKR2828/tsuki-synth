"""WF0925-P1: parse both workflows with PyYAML, check the structural facts
this card changed, and simulate the new HostProbe step's pwsh script locally
the way GitHub's `shell: pwsh` runs it (prepend $ErrorActionPreference =
'stop', append the LASTEXITCODE exit), against STUB probes (no real
HostProbe run -> nothing written to %APPDATA%)."""
import os
import re
import subprocess
import sys
import tempfile
from pathlib import Path

import yaml

ROOT = Path(r"<repo>")
WF = {
    "release-physics.yml": ROOT / ".github" / "workflows" / "release-physics.yml",
    "physics.yml": ROOT / ".github" / "workflows" / "physics.yml",
}
fails = 0


def check(cond, msg):
    global fails
    print(("  PASS  " if cond else "  FAIL  ") + msg)
    if not cond:
        fails += 1


docs = {}
for name, path in WF.items():
    with open(path, encoding="utf-8") as f:
        doc = yaml.safe_load(f)
    docs[name] = doc
    jobs = doc.get("jobs", {})
    n_steps = sum(len(j.get("steps", [])) for j in jobs.values())
    # PyYAML (YAML 1.1) reads the bare key `on` as boolean True.
    trig = doc.get("on", doc.get(True))
    print("%s: yaml.safe_load OK; jobs=%s; steps=%d; triggers=%s"
          % (name, list(jobs), n_steps, list(trig) if isinstance(trig, dict) else trig))


def step(job, name_prefix):
    for s in job["steps"]:
        if s.get("name", "").startswith(name_prefix):
            return s
    return None


def index_of(job, name_prefix):
    for i, s in enumerate(job["steps"]):
        if s.get("name", "").startswith(name_prefix):
            return i
    return -1


print("\n[release-physics.yml build-and-host-validate]")
rj = docs["release-physics.yml"]["jobs"]["build-and-host-validate"]
build = step(rj, "Configure and build all release targets")["run"]
for tgt in ("TsukiSynthAuditTest", "TsukiSynthTunerTest", "TsukiSynthPhysicsModelsTest",
            "TsukiSynthSpectrumViewTest", "TsukiSynthHostProbe", "TsukiSynth_VST3"):
    check(tgt in build.split(), "build target list contains %s" % tgt)
gates = step(rj, "Run C++ and Python physics gates")["run"]
check("python -m pytest tests -q" in gates, "gates step runs `python -m pytest tests -q`")
check("unittest" not in gates, "gates step no longer uses unittest")
i_install = index_of(rj, "Install Python dependencies")
install = rj["steps"][i_install]["run"]
check("-r tools/requirements-physics.txt" in install and i_install < index_of(rj, "Run C++ and Python physics gates"),
      "requirements (pins pytest==8.4.2) installed before the pytest step")
req = (ROOT / "tools" / "requirements-physics.txt").read_text(encoding="utf-8")
check("pytest==" in req, "tools/requirements-physics.txt pins pytest (%s)"
      % [l for l in req.splitlines() if l.startswith("pytest")])
hp = step(rj, "Run L2 HostProbe")
check(hp is not None, "HostProbe run step present")
check(hp.get("working-directory") == "${{ github.workspace }}", "HostProbe step cwd = repo root (${{ github.workspace }})")
check(hp["env"]["HOSTPROBE_VST3"] == "build/TsukiSynth_artefacts/Release/VST3/TsukiSynth.vst3",
      "HostProbe points at the VST3 this job built")
check(index_of(rj, "Configure and build all release targets") < index_of(rj, "Run L2 HostProbe"),
      "HostProbe step runs after the build step")
up = step(rj, "Upload HostProbe log")
check(up is not None and up.get("if") == "always()", "HostProbe log uploaded with if: always()")

print("\n[physics.yml build-and-verify]")
pj = docs["physics.yml"]["jobs"]["build-and-verify"]
tb = step(pj, "Build regression tests")["run"]
for tgt in ("TsukiSynthAuditTest", "TsukiSynthTunerTest", "TsukiSynthPhysicsModelsTest",
            "TsukiSynthSpectrumViewTest", "TsukiSynthHostProbe"):
    check(tgt in tb.split(), "test build target list contains %s" % tgt)
hp2 = step(pj, "Run L2 HostProbe")
check(hp2 is not None, "HostProbe run step present")
check(hp2.get("working-directory") == "${{ github.workspace }}", "HostProbe step cwd = repo root")
check(index_of(pj, "Build VST3") < index_of(pj, "Run L2 HostProbe")
      and index_of(pj, "Build regression tests") < index_of(pj, "Run L2 HostProbe"),
      "HostProbe step runs after VST3 + HostProbe are built")
up2 = step(pj, "Upload HostProbe log")
check(up2 is not None and up2.get("if") == "always()", "HostProbe log uploaded with if: always()")
check(hp["run"] == hp2["run"], "both workflows use the identical HostProbe script")
xp = docs["physics.yml"]["jobs"]["cross-platform-emit"]["strategy"]["matrix"]["include"]
check([m["label"] for m in xp] == ["windows-2022-msvc", "ubuntu-24.04-clang", "macos-14-appleclang"],
      "cross-platform labels unchanged (Linux label left for 月月's ruling)")
asan = step(docs["physics.yml"]["jobs"]["cpp-address-sanitizer"], "Build AddressSanitizer")["run"]
check("TsukiSynthHostProbe" not in asan, "ASan job untouched (HostProbe not added there)")

# ---- simulate the pwsh step -------------------------------------------------
print("\n[local simulation of the HostProbe step script, GitHub pwsh wrapper, stub probes]")
script = hp["run"]
wrapper = ("$ErrorActionPreference = 'stop'\n" + script +
           "\nif ((Test-Path -LiteralPath variable:\\LASTEXITCODE)) { exit $LASTEXITCODE }\n")
with tempfile.TemporaryDirectory(prefix="p1_hp_sim_") as td:
    td = Path(td)
    ws = td / "workspace"
    (ws / "build" / "Release").mkdir(parents=True)
    (ws / "build" / "TsukiSynth_artefacts" / "Release" / "VST3" / "TsukiSynth.vst3").mkdir(parents=True)
    ps1 = td / "step.ps1"
    ps1.write_text(wrapper, encoding="utf-8")

    def stub(name, code, text):
        p = td / name
        p.write_text("@echo off\r\necho stub cwd=%%CD%%\r\necho stub args=%%*\r\necho %s\r\nexit /b %d\r\n"
                     % (text, code), encoding="ascii")
        return p

    ok_stub = stub("probe_ok.cmd", 0, "PASS (0 failures)")
    bad_stub = stub("probe_fail.cmd", 1, "FAIL (1 failures)")
    cases = [
        ("probe exits 0", str(ok_stub), "build/TsukiSynth_artefacts/Release/VST3/TsukiSynth.vst3", 0),
        ("probe exits 1", str(bad_stub), "build/TsukiSynth_artefacts/Release/VST3/TsukiSynth.vst3", 1),
        ("probe binary missing", "build/Release/TsukiSynthHostProbe.exe",
         "build/TsukiSynth_artefacts/Release/VST3/TsukiSynth.vst3", "nonzero"),
        ("VST3 bundle missing", str(ok_stub), "build/nope/TsukiSynth.vst3", "nonzero"),
    ]
    for label, exe, vst3, want in cases:
        out = td / ("out_" + label.replace(" ", "_"))
        env = dict(os.environ, HOSTPROBE_EXE=exe, HOSTPROBE_VST3=vst3, HOSTPROBE_OUT=str(out))
        r = subprocess.run(["pwsh", "-NoProfile", "-command", ". '%s'" % ps1], cwd=ws, env=env,
                           capture_output=True, text=True)
        log = out / "hostprobe_log.txt"
        good = (r.returncode == want) if want != "nonzero" else (r.returncode != 0)
        check(good, "%-22s -> exit %d (want %s)" % (label, r.returncode, want))
        tail = [l for l in (r.stdout + r.stderr).splitlines() if l.strip()][-4:]
        for l in tail:
            print("          | " + re.sub(r"\[[0-9;]*m", "", l)[:160])
        if want in (0, 1):
            txt = log.read_text(encoding="utf-8", errors="replace") if log.exists() else ""
            check(("PASS (0 failures)" if want == 0 else "FAIL (1 failures)") in txt,
                  "%-22s -> log file captured the probe's verdict line" % label)
            check(("stub cwd=%s" % ws) in txt, "%-22s -> probe ran with cwd = workspace root" % label)

print("\nRESULT: %s (%d failed checks)" % ("ALL PASS" if fails == 0 else "FAILURES", fails))
sys.exit(1 if fails else 0)
