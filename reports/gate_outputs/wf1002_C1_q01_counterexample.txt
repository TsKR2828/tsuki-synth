# 作法：<SCRATCH>/c1/q01_counter/ 是獨立 CMake 專案：src/ 是 repo src/ 的複本（只改 EffectChain.h），
#       tests/audit_repro.cpp 與 repo 版逐字相同，JUCE 直接用 <REPO>/libs/JUCE。建置在 <SCRATCH>/c1/q01_counter/build。

## scratch CMakeLists.txt
# WF1002-C1 Q01 counter-example scratch project (NOT part of the repo).
# Builds tests/audit_repro.cpp against a scratch COPY of src/ whose
# EffectChain.h is modified per variant; JUCE comes from the repo's libs/JUCE.
cmake_minimum_required(VERSION 3.22)
project(Q01Counter VERSION 0.0.1)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_compile_options(/utf-8)
endif()
add_subdirectory(<REPO>/libs/JUCE ${CMAKE_BINARY_DIR}/JUCE)
add_executable(AuditCounter tests/audit_repro.cpp)
target_include_directories(AuditCounter PRIVATE src)
target_compile_definitions(AuditCounter PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(AuditCounter PRIVATE juce::juce_audio_formats juce::juce_audio_processors juce::juce_dsp
                      PUBLIC juce::juce_recommended_config_flags)
