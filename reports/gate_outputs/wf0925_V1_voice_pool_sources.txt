  525  SynthesiserVoice* Synthesiser::findVoiceToSteal (SynthesiserSound* soundToPlay,
  526                                                   int /*midiChannel*/, int midiNoteNumber) const
  527  {
  528      // This voice-stealing algorithm applies the following heuristics:
  529      // - Re-use the oldest notes first
  530      // - Protect the lowest & topmost notes, even if sustained, but not if they've been released.
  531  
  532      // apparently you are trying to render audio without having any voices...
  533      jassert (! voices.isEmpty());
  534  
  535      // These are the voices we want to protect (ie: only steal if unavoidable)
  536      SynthesiserVoice* low = nullptr; // Lowest sounding note, might be sustained, but NOT in release phase
  537      SynthesiserVoice* top = nullptr; // Highest sounding note, might be sustained, but NOT in release phase
  538  
  539      // All major OSes use double-locking so this will be lock- and wait-free as long as the lock is not
  540      // contended. This is always the case if you do not call findVoiceToSteal on multiple threads at
  541      // the same time.
  542      const ScopedLock sl (stealLock);
  543  
  544      // this is a list of voices we can steal, sorted by how long they've been running
  545      usableVoicesToStealArray.clear();
  546  
  547      for (auto* voice : voices)
  548      {
  549          if (voice->canPlaySound (soundToPlay))
  550          {
  551              jassert (voice->isVoiceActive()); // We wouldn't be here otherwise
  552  
  553              usableVoicesToStealArray.add (voice);
  554  
  555              // NB: Using a functor rather than a lambda here due to scare-stories about
  556              // compilers generating code containing heap allocations.
  557              struct Sorter
  558              {
  559                  bool operator() (const SynthesiserVoice* a, const SynthesiserVoice* b) const noexcept { return a->wasStartedBefore (*b); }
  560              };
  561  
  562              std::sort (usableVoicesToStealArray.begin(), usableVoicesToStealArray.end(), Sorter());
  563  
  564              if (! voice->isPlayingButReleased()) // Don't protect released notes
  565              {
  566                  auto note = voice->getCurrentlyPlayingNote();
  567  
  568                  if (low == nullptr || note < low->getCurrentlyPlayingNote())
  569                      low = voice;
  570  
  571                  if (top == nullptr || note > top->getCurrentlyPlayingNote())
  572                      top = voice;
  573              }
  574          }
  575      }
  576  
  577      // Eliminate pathological cases (ie: only 1 note playing): we always give precedence to the lowest note(s)
  578      if (top == low)
  579          top = nullptr;
  580  
  581      // The oldest note that's playing with the target pitch is ideal.
  582      for (auto* voice : usableVoicesToStealArray)
  583          if (voice->getCurrentlyPlayingNote() == midiNoteNumber)
  584              return voice;
  585  
  586      // Oldest voice that has been released (no finger on it and not held by sustain pedal)
  587      for (auto* voice : usableVoicesToStealArray)
  588          if (voice != low && voice != top && voice->isPlayingButReleased())
  589              return voice;
  590  
  591      // Oldest voice that doesn't have a finger on it:
  592      for (auto* voice : usableVoicesToStealArray)
  593          if (voice != low && voice != top && ! voice->isKeyDown())
  594              return voice;
  595  
  596      // Oldest voice that isn't protected
  597      for (auto* voice : usableVoicesToStealArray)
  598          if (voice != low && voice != top)
  599              return voice;
  600  
  601      // We've only got "protected" voices now: lowest note takes priority
  602      jassert (low != nullptr);
  603  
  604      // Duophonic synth: give priority to the bass note:
  605      if (top != nullptr)
  606          return top;
  607  
  608      return low;
  609  }
