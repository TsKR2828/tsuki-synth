import json
for label, path in [('weak23', 'output/wf0908/P1/partial_verify_weak_report.json'),
                     ('general40', 'output/wf0908/P1/partial_verify_general40_report.json')]:
    d = json.load(open(path, encoding='utf-8'))
    counts = {}
    max_abs = 0.0

            c = p.get('measured_cents')
