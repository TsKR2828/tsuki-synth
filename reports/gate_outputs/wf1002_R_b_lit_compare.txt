def t60_from_rate(r): return 60.0/abs(r) if r < 0 else float("inf")

print("note | model 0.8mm / 1.0mm / cand | Wogram 3xT20 (fig, median +-2 st) | Cheng partial-1 T60 f/m/p (median +-2 st of 60/|rate|) | Lehtonen07 fund. T60 (6-s fit) | Lehtonen09 initial T60")

    for dyn in ("f","m","p"):
