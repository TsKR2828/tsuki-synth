// WF1002-C2 (R10 before/after card) -- ISOLATED-COPY-ONLY measurement harness.
// Lives only in output/wf1002/C2/tree* (git-ignored copies of HEAD b41298c);
// NOT part of the product, NOT a GATE. Every number it prints is descriptive.
//
// Two modes:
//
//  vst3pair <base.vst3> <fix.vst3> <events.txt> <out_prefix>
//      Loads BOTH built TsukiSynth.vst3 bundles (the binaries a DAW loads)
//      through juce::VST3PluginFormat, selects Engine = "FM Piano", applies
//      the per-note FM parameter sets of <events.txt> as block-start
//      automation, sets Reverb Mix = 0 (HostProbe renderNotes() convention;
//      every other parameter stays at its default), streams the same MIDI
//      into both in lock-step (512-sample blocks, 48 kHz, non-realtime) and
//      writes per-block statistics (sum of squares, peak, max |base-fix|)
//      plus wall-clock processBlock time per instance.
//
//  mirror <numVoices> <events.txt> <out_prefix>
//      A juce::Synthesiser with <numVoices> FMPianoVoice built from THIS
//      tree's src/engines/FMPianoEngine.h + src/dsp/Envelope.h, wired the
//      way src/PluginProcessor.cpp wires the FM pool (same Sound, same
//      setNoiseIdentity(i), same parameter atomics, Macros at their defaults,
//      same single renderNextBlock call per block) -- the VST3 is a black box
//      for voice counts, so voice occupancy, steals and envelope levels are
//      read here. 16 = the plug-in's pool; a large number = "unlimited pool"
//      demand (fm_zombie_check method). No effect chain (dry voice sum).
//
// events.txt (written by c2_make_events.py):
//   sr <hz> | block <n> | len <samples>
//   pset <id> <type> <ratio> <index> <brightness> <feedback> <attack_ms> <release_ms>
//   ev <sample> <on|off|pdn|pup> <midi> <velocity> <pset>   (pre-sorted)

struct PSet { int type; float ratio, index, brightness, feedback, attackMs, releaseMs; };
struct Ev   { long long sample; int kind; int midi; float vel; int pset; };   // kind 0 off,1 on,2 pdn,3 pup

struct Events
{
    double sr = 48000.0; int block = 512; long long len = 0;
    std::map<int, PSet> psets;
    std::vector<Ev> evs;
};

Events loadEvents (const std::string& path)
{
    Events E;
    std::ifstream f (path);
    std::string line;
    while (std::getline (f, line))
    {
        std::istringstream ss (line);
        std::string tag; ss >> tag;
        if (tag == "sr") ss >> E.sr;
        else if (tag == "block") ss >> E.block;
        else if (tag == "len") ss >> E.len;
        else if (tag == "pset")
        {
            int id; PSet p; ss >> id >> p.type >> p.ratio >> p.index >> p.brightness
                               >> p.feedback >> p.attackMs >> p.releaseMs;
            E.psets[id] = p;
        }
        else if (tag == "ev")
        {
            Ev e; std::string k; ss >> e.sample >> k >> e.midi >> e.vel >> e.pset;
            e.kind = k == "off" ? 0 : k == "on" ? 1 : k == "pdn" ? 2 : 3;
            E.evs.push_back (e);
        }
    }
    return E;
}

juce::MidiMessage toMidi (const Ev& e)
{
    switch (e.kind)
    {
        case 0:  return juce::MidiMessage::noteOff (1, e.midi);
        case 1:  return juce::MidiMessage::noteOn  (1, e.midi, e.vel);
        case 2:  return juce::MidiMessage::controllerEvent (1, 64, 127);
        default: return juce::MidiMessage::controllerEvent (1, 64, 0);
    }
}

// Walks the event list block by block: fills the MIDI for the block starting
// at blockStart and returns the pset that must be active at the block start
// (the pset of the first note-on in the block, -1 if none). Counts blocks
// holding note-ons of more than one pset (those notes then share the first).
struct Streamer
{
    const Events& E; size_t next = 0; int mixedBlocks = 0;
    explicit Streamer (const Events& e) : E (e) {}
    int fill (long long blockStart, juce::MidiBuffer& midi)
    {
        midi.clear();
        int want = -1; bool mixed = false;
        while (next < E.evs.size() && E.evs[next].sample < blockStart + E.block)
        {
            const auto& e = E.evs[next++];
            midi.addEvent (toMidi (e), (int) (e.sample - blockStart));
            if (e.kind == 1)
            {
                if (want < 0) want = e.pset;
                else if (want != e.pset) mixed = true;
            }
        }
        if (mixed) ++mixedBlocks;
        return want;
    }
};

juce::AudioProcessorParameter* exactParam (juce::AudioPluginInstance& inst, const juce::String& name)
{
    for (auto* p : inst.getParameters())
        if (p->getName (128) == name) return p;
    return nullptr;
}

bool setByText (juce::AudioPluginInstance& inst, const juce::String& name, const juce::String& text,
                std::ostream& log)
{
    auto* p = exactParam (inst, name);
    if (p == nullptr) { log << "  !! parameter not found: " << name << "\n"; return false; }
    p->setValue (p->getValueForText (text));
    log << "  set " << name << " = " << text << " -> '" << p->getCurrentValueAsText() << "'\n";
    return true;
}

void applyPSet (juce::AudioPluginInstance& inst, const PSet& p, std::ostream& log)
{
    static const char* types[] = { "Piano", "E.Piano", "Vibraphone", "Bell", "Organ", "Pad", "Bass", "Brass" };
    setByText (inst, "Sound Type", types[juce::jlimit (0, 7, p.type)], log);
    setByText (inst, "FM Ratio",        juce::String (p.ratio, 6), log);
    setByText (inst, "Mod Index",       juce::String (p.index, 6), log);
    setByText (inst, "Tone Decay",      juce::String (p.brightness, 6), log);
    setByText (inst, "Feedback",        juce::String (p.feedback, 6), log);
    setByText (inst, "FM Attack (ms)",  juce::String (p.attackMs, 6), log);
    setByText (inst, "FM Release (ms)", juce::String (p.releaseMs, 6), log);
}

int runPair (const juce::String& baseBundle, const juce::String& fixBundle,
             const std::string& evPath, const std::string& prefix)
{
    const Events E = loadEvents (evPath);
    std::ofstream log (prefix + "_log.txt");
    juce::AudioPluginFormatManager fm;
    fm.addFormat (new juce::VST3PluginFormat());
    juce::VST3PluginFormat vst3;

    std::unique_ptr<juce::AudioPluginInstance> inst[2];
    const juce::String bundles[2] = { baseBundle, fixBundle };
    for (int k = 0; k < 2; ++k)
    {
        juce::OwnedArray<juce::PluginDescription> found;
        vst3.findAllTypesForFile (found, bundles[k]);
        if (found.isEmpty()) { std::cerr << "no plugin in " << bundles[k] << "\n"; return 2; }
        juce::String err;
        inst[k] = fm.createPluginInstance (*found[0], E.sr, E.block, err);
        if (inst[k] == nullptr) { std::cerr << "create failed: " << err << "\n"; return 2; }
        log << "instance " << k << ": " << bundles[k] << " (" << found[0]->name << " "
            << found[0]->version << ")\n";
        setByText (*inst[k], "Engine", "FM Piano", log);
        setByText (*inst[k], "Reverb Mix", "0", log);
        inst[k]->setNonRealtime (true);
        inst[k]->prepareToPlay (E.sr, E.block);
    }

    const long long nBlocks = (E.len + E.block - 1) / E.block;
    std::ofstream csv (prefix + "_blocks.csv");
    csv << "block,base_sumsq,fix_sumsq,base_peak,fix_peak,diff_sumsq,diff_maxabs\n";
    Streamer s0 (E), s1 (E);
    juce::MidiBuffer m0, m1;
    const int chans = juce::jmax (2, inst[0]->getTotalNumOutputChannels());
    juce::AudioBuffer<float> b0 (chans, E.block), b1 (chans, E.block);
    int cur[2] = { -1, -1 };
    double secs[2] = { 0.0, 0.0 };
    for (long long b = 0; b < nBlocks; ++b)
    {
        const long long bs = b * E.block;
        int w0 = s0.fill (bs, m0), w1 = s1.fill (bs, m1);
        if (w0 >= 0 && w0 != cur[0]) { applyPSet (*inst[0], E.psets.at (w0), log); cur[0] = w0; }
        if (w1 >= 0 && w1 != cur[1]) { applyPSet (*inst[1], E.psets.at (w1), log); cur[1] = w1; }
        b0.clear(); b1.clear();
        auto t0 = std::chrono::steady_clock::now();
        inst[0]->processBlock (b0, m0);
        auto t1 = std::chrono::steady_clock::now();
        inst[1]->processBlock (b1, m1);
        auto t2 = std::chrono::steady_clock::now();
        secs[0] += std::chrono::duration<double> (t1 - t0).count();
        secs[1] += std::chrono::duration<double> (t2 - t1).count();
        double sq0 = 0, sq1 = 0, sqd = 0; float pk0 = 0, pk1 = 0, mx = 0;
        for (int c = 0; c < 2; ++c)
        {
            const float* x0 = b0.getReadPointer (juce::jmin (c, chans - 1));
            const float* x1 = b1.getReadPointer (juce::jmin (c, chans - 1));
            for (int i = 0; i < E.block; ++i)
            {
                sq0 += (double) x0[i] * x0[i]; sq1 += (double) x1[i] * x1[i];
                const float d = x0[i] - x1[i]; sqd += (double) d * d;
                pk0 = std::max (pk0, std::abs (x0[i])); pk1 = std::max (pk1, std::abs (x1[i]));
                mx = std::max (mx, std::abs (d));
            }
        }
        csv << b << ',' << sq0 << ',' << sq1 << ',' << pk0 << ',' << pk1 << ',' << sqd << ',' << mx << '\n';
    }
    for (int k = 0; k < 2; ++k) inst[k]->releaseResources();
    const double audioSecs = (double) (nBlocks * E.block) / E.sr;
    std::ofstream sum (prefix + "_summary.txt");
    sum << "blocks " << nBlocks << " block_size " << E.block << " sr " << E.sr
        << " audio_s " << audioSecs << "\n"
        << "mixed_pset_blocks " << s0.mixedBlocks << "\n"
        << "process_s base " << secs[0] << " fix " << secs[1] << "\n"
        << "realtime_fraction base " << secs[0] / audioSecs << " fix " << secs[1] / audioSecs << "\n";
    std::cout << "pair done: " << nBlocks << " blocks, base " << secs[0] << " s, fix " << secs[1] << " s\n";
    return 0;
}

// ------------------------------------------------------------------ mirror
class MirrorSynth : public juce::Synthesiser
{
public:
    std::vector<long long> onSamples;   // k-th note-on handled <-> k-th "on" event
    size_t onIdx = 0;
    long long curSample = 0;
    long long steals = 0, stealsKeyDown = 0, stealsReleased = 0;
    long long releaseRetriggers = 0;  // stopVoice(tail-off) hitting a voice already in Release
    long long tailOffHits = 0;        // every stopVoice(tail-off) on a voice of that note
    std::vector<std::string> stealLog;
    bool pedal = false;
    std::function<void()> afterNoteOn;

    static FMPianoVoice* fm (juce::SynthesiserVoice* v) { return static_cast<FMPianoVoice*> (v); }

    void countRetrig (int note, bool isNoteOff)
    {
        if (isNoteOff && pedal) return;   // CC64 held: JUCE noteOff defers the tail-off
        for (int i = 0; i < getNumVoices(); ++i)
        {
            auto* v = getVoice (i);
            if (v->getCurrentlyPlayingNote() == note)
            {
                ++tailOffHits;
                if (fm (v)->ampEnv.getState() == Envelope::State::Release) ++releaseRetriggers;
            }
        }
    }
    void noteOn (int ch, int note, float vel) override
    {
        curSample = onIdx < onSamples.size() ? onSamples[onIdx] : -1;
        ++onIdx;
        countRetrig (note, false);
        juce::Synthesiser::noteOn (ch, note, vel);
        if (afterNoteOn) afterNoteOn();
    }
    void noteOff (int ch, int note, float vel, bool tail) override
    {
        countRetrig (note, true);
        juce::Synthesiser::noteOff (ch, note, vel, tail);
    }
    void handleSustainPedal (int ch, bool down) override
    {
        pedal = down;
        juce::Synthesiser::handleSustainPedal (ch, down);
    }
    juce::SynthesiserVoice* findFreeVoice (juce::SynthesiserSound* s, int ch, int note, bool steal) const override
    {
        auto* v = juce::Synthesiser::findFreeVoice (s, ch, note, steal);
        if (v != nullptr && v->isVoiceActive())
        {
            auto* self = const_cast<MirrorSynth*> (this);
            ++self->steals;
            const bool kd = v->isKeyDown();
            const auto st = fm (v)->ampEnv.getState();
            if (kd) ++self->stealsKeyDown;
            if (st == Envelope::State::Release) ++self->stealsReleased;
            const float lv = fm (v)->ampEnv.currentLevel;
            std::ostringstream o;
            o << curSample << ',' << note << ',' << v->getCurrentlyPlayingNote() << ',' << (kd ? 1 : 0)
              << ',' << (int) st << ',' << (lv > 0 ? 20.0 * std::log10 ((double) lv) : -999.0);
            self->stealLog.push_back (o.str());
        }
        return v;
    }
};

int runMirror (int numVoices, const std::string& evPath, const std::string& prefix)
{
    const Events E = loadEvents (evPath);
    // initial values = APVTS defaults (ParameterLayout.cpp); overwritten per pset
    std::atomic<float> aType { 0 }, aRatio { 1.0f }, aIndex { 4.5f }, aBright { 0.6f }, aFb { 0.02f },
                       aAtt { 5.0f }, aRel { 500.0f };
    std::atomic<float> mM { 0.5f }, mT { 0.5f }, mD { 0.5f }, mS { 0.5f }, mB { 0.5f }, mY { 0.5f }, mN { 0.0f };
    MirrorSynth synth;
    synth.addSound (new FMPianoSound());
    for (int i = 0; i < numVoices; ++i)
    {
        auto* v = new FMPianoVoice();
        v->setNoiseIdentity ((uint64_t) i);
        v->pType = &aType; v->pRatio = &aRatio; v->pIndex = &aIndex; v->pBrightness = &aBright;
        v->pFeedback = &aFb; v->pAttack = &aAtt; v->pRelease = &aRel;
        v->pMacroMaterial = &mM; v->pMacroTension = &mT; v->pMacroDamping = &mD; v->pMacroStrike = &mS;
        v->pMacroBrightness = &mB; v->pMacroBody = &mY; v->pMacroNoise = &mN;
        synth.addVoice (v);
    }
    synth.setCurrentPlaybackSampleRate (E.sr);
    for (auto& e : E.evs) if (e.kind == 1) synth.onSamples.push_back (e.sample);

    auto activeCount = [&]
    {
        int n = 0;
        for (int i = 0; i < synth.getNumVoices(); ++i) n += synth.getVoice (i)->isVoiceActive() ? 1 : 0;
        return n;
    };

    int peak = 0, maxHere = 0; long long peakSample = -1;
    std::map<std::string, int> peakBins; int peakKeyDown = 0; std::map<int, int> peakNotes;
    synth.afterNoteOn = [&]
    {
        const int n = activeCount();
        maxHere = std::max (maxHere, n);
        if (n <= peak) return;
        peak = n; peakSample = synth.curSample;
        peakBins.clear(); peakNotes.clear(); peakKeyDown = 0;
        for (int i = 0; i < synth.getNumVoices(); ++i)
        {
            auto* v = synth.getVoice (i);
            if (! v->isVoiceActive()) continue;
            const float lv = MirrorSynth::fm (v)->ampEnv.currentLevel;
            const double db = lv > 0 ? 20.0 * std::log10 ((double) lv) : -1e9;
            std::string bin = lv <= 0 ? "level 0" : db > -10 ? "(-10,0]" : db > -60 ? "(-60,-10]"
                            : db > -138.5 ? "(-138.5,-60]" : "<=-138.5";
            ++peakBins[bin];
            if (v->isKeyDown()) ++peakKeyDown;
            ++peakNotes[v->getCurrentlyPlayingNote()];
        }
    };

    const long long nBlocks = (E.len + E.block - 1) / E.block;
    std::ofstream csv (prefix + "_voices.csv");
    csv << "block,active_end,active_max_at_noteons,sumsq,peak\n";
    juce::AudioBuffer<float> buf (2, E.block);
    juce::MidiBuffer midi;
    Streamer streamer (E);
    long long voiceBlocks = 0;
    double secs = 0.0;
    int cur = -1;
    auto applyP = [&] (const PSet& p)
    {
        aType = (float) p.type; aRatio = p.ratio; aIndex = p.index; aBright = p.brightness;
        aFb = p.feedback; aAtt = p.attackMs; aRel = p.releaseMs;
    };
    for (long long b = 0; b < nBlocks; ++b)
    {
        juce::ScopedNoDenormals noDenormals;   // PluginProcessor::processBlock does the same
        const long long bs = b * E.block;
        buf.clear();
        // Same call the plug-in makes (PluginProcessor::processBlock): the
        // whole block's MIDI in one renderNextBlock, so JUCE's own sub-block
        // split (minimumSubBlockSize 32, non-strict) applies unchanged.
        int want = streamer.fill (bs, midi);
        if (want >= 0 && want != cur) { applyP (E.psets.at (want)); cur = want; }
        maxHere = 0;
        auto t0 = std::chrono::steady_clock::now();
        synth.renderNextBlock (buf, midi, 0, E.block);
        secs += std::chrono::duration<double> (std::chrono::steady_clock::now() - t0).count();
        const int endN = activeCount();
        voiceBlocks += endN;
        double sq = 0; float pk = 0;
        const float* x = buf.getReadPointer (0);
        for (int i = 0; i < E.block; ++i) { sq += (double) x[i] * x[i]; pk = std::max (pk, std::abs (x[i])); }
        csv << b << ',' << endN << ',' << std::max (maxHere, endN) << ',' << sq << ',' << pk << '\n';
    }
    std::ofstream st (prefix + "_steals.csv");
    st << "sample,incoming_note,victim_note,victim_key_down,victim_env_state,victim_env_db\n";
    for (auto& l : synth.stealLog) st << l << '\n';
    std::ofstream sum (prefix + "_summary.txt");
    const double audioSecs = (double) (nBlocks * E.block) / E.sr;
    sum << "voices " << numVoices << " blocks " << nBlocks << " audio_s " << audioSecs << "\n"
        << "mixed_pset_blocks " << streamer.mixedBlocks << "\n"
        << "note_ons_handled " << synth.onIdx << " of " << synth.onSamples.size() << "\n"
        << "peak_active " << peak << " at_sample " << peakSample << " t_s " << (double) peakSample / E.sr << "\n"
        << "peak_key_down " << peakKeyDown << " released " << (peak - peakKeyDown) << "\n"
        << "peak_env_bins";
    for (auto& kv : peakBins) sum << " [" << kv.first << "]=" << kv.second;
    sum << "\npeak_notes";
    std::vector<std::pair<int,int>> pn (peakNotes.begin(), peakNotes.end());
    std::sort (pn.begin(), pn.end(), [] (auto& a, auto& c) { return a.second > c.second; });
    for (size_t i = 0; i < std::min<size_t> (10, pn.size()); ++i) sum << " " << pn[i].first << ":" << pn[i].second;
    sum << "\nmean_active_at_block_end " << (double) voiceBlocks / (double) nBlocks << "\n"
        << "steals " << synth.steals << " victim_key_down " << synth.stealsKeyDown
        << " victim_in_release " << synth.stealsReleased << "\n"
        << "tail_off_hits_same_note " << synth.tailOffHits
        << " of_which_voice_already_in_release " << synth.releaseRetriggers << "\n"
        << "render_s " << secs << " realtime_fraction " << secs / audioSecs << "\n";
    std::cout << "mirror done: voices " << numVoices << " peak " << peak << " steals " << synth.steals << "\n";
    return 0;
}
} // namespace

int main (int argc, char** argv)
{
    juce::ScopedJuceInitialiser_GUI juceInit;
    juce::MessageManager::getInstance()->setCurrentThreadAsMessageThread();
    const std::string mode = argc > 1 ? argv[1] : "";
    if (mode == "vst3pair" && argc == 6)
        return runPair (juce::String (argv[2]), juce::String (argv[3]), argv[4], argv[5]);
    if (mode == "mirror" && argc == 5)
        return runMirror (std::atoi (argv[2]), argv[3], argv[4]);
    std::cerr << "usage: c2_fm_harness vst3pair <base.vst3> <fix.vst3> <events.txt> <prefix>\n"
                 "       c2_fm_harness mirror <numVoices> <events.txt> <prefix>\n";
    return 2;
}

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""WF1002-C2 helper (isolated copy, descriptive only): write the event files
c2_fm_harness reads.

Corpus scores: the FM events of a score, extracted with the SAME code the
WF0925-V1 voice-pool model uses (reports/wf0925_method/voice_pool_occupancy.py
load_events + build_messages conventions): note-on at round(time*sr),
note-off at round((time+duration)*sr) (scenario A = HostProbe renderNotes()
convention), offs before ons at the same sample. Scenario C = CC64 held from
sample 0 until one sample after the last note-off (the V1 report's
hypothetical "whole piece pedal" upper bound). Per-event FM params from the
score (fm_preset/ratio/index/brightness/feedback/attack/release; absent keys
= FMParams defaults = APVTS defaults) become parameter sets that the harness
applies as block-start automation.

Usage: python c2_make_events.py --repo <tsuki-synth> --dumps <dir> --out-dir <dir>
"""
import argparse
import json
import sys
from pathlib import Path

DEF = {"fm_preset": 0, "fm_ratio": 1.0, "fm_index": 4.5, "fm_brightness": 0.6,
       "fm_feedback": 0.02, "fm_attack": 5.0, "fm_release": 500.0}
SR = 48000
BLOCK = 512


def pset_of(params):
    p = dict(DEF)
    p.update({k: v for k, v in (params or {}).items() if k in DEF})
    return (int(p["fm_preset"]), float(p["fm_ratio"]), float(p["fm_index"]),
            float(p["fm_brightness"]), float(p["fm_feedback"]), float(p["fm_attack"]),
            float(p["fm_release"]))


def write(out, notes, pedal=False, tail_s=2.0):
    """notes: list of (t_on_s, dur_s, midi, vel, pset_tuple)."""
    psets, rows = {}, []
    for order, (t, d, m, v, ps) in enumerate(notes):
        pid = psets.setdefault(ps, len(psets))
        on = int(round(t * SR))
        off = max(on, int(round((t + max(0.0, d)) * SR)))
        rows.append((on, 1, order, "on", m, v, pid))
        rows.append((off, 0, order, "off", m, 0.0, pid))
    last_off = max(r[0] for r in rows if r[3] == "off")

        rows.append((0, -1, -1, "pdn", 0, 0.0, 0))
        rows.append((last_off + 1, -1, -1, "pup", 0, 0.0, 0))
    rows.sort(key=lambda r: (r[0], r[1], r[2]))
    max_rel = max(ps[6] for ps in psets) / 1000.0
    length = last_off + 2 + int((max_rel + tail_s) * SR)
    with open(out, "w", encoding="utf-8", newline="\n") as f:
        f.write("sr %d\nblock %d\nlen %d\n" % (SR, BLOCK, length))
        for ps, pid in psets.items():

#!/usr/bin/env python3

"""WF1002-C2: re-run the WF0925-V1 fm_zombie_check.py method twice --
once with the model's own (HEAD-code) rule, once with the C2 fix rule --
on scores/examples/moonlight_sonata_complete.score.json. Descriptive only.

The ONLY difference between the two runs is voice_pool_occupancy.stop_tail_off
for FM voices:
  base  (vpo as shipped): every tail-off restarts the release ramp from the
        current level over the full release time (Envelope::noteOff at HEAD).
  fix   (C2 patch): a voice already releasing keeps its ramp; only the first
        tail-off starts it. (The patch also lets a SHORTER release -- the 5 ms
        hard-steal ramp -- take over; the model never issues hard-steal
        tail-offs with an unlimited pool, so that branch is not exercised.)
Everything else (events, scenario A, unlimited pool, bins, 16-pool steal
rules) is vpo / fm_zombie_check code, imported, not re-implemented.

Usage: python c2_fm_zombie_fix_model.py <repo> <dumps>

from pathlib import Path

REPO = Path(sys.argv[1]).resolve()
sys.path.insert(0, str(REPO / "reports" / "wf0925_method"))
import voice_pool_occupancy as vpo  # noqa: E402

_orig_stop = vpo.stop_tail_off


def stop_tail_off_fix(v, t):
    if v.ev["family"] == "fm" and v.fm_rel_t is not None:
        return                      # already in Release: keep the running ramp
    _orig_stop(v, t)


def run(events, sr, label):
    r = vpo.simulate(events, sr, "A", None)
    t = r["max_t"]
    n_peak = int(round(t * sr))
    r2 = vpo.simulate(events, sr, "A", None, until_sample=n_peak)
    live = [v for v in r2["voices"] if v.end * sr > n_peak + 1e-9]
    bins, notes = Counter(), Counter()
    for v in live:
        lvl = vpo.fm_level(v, t)
        db = 20 * math.log10(lvl) if lvl > 0 else None
        if db is None:
            b = "level 0"
        elif db > -10:
            b = "(-10,0]"
        elif db > -60:
            b = "(-60,-10]"
        elif db > -138.5:
            b = "(-138.5,-60]"
        else:
            b = "<=-138.5"
        bins[b] += 1
        notes[(v.ev["note"], v.key_down)] += 1
    r16 = vpo.simulate(events, sr, "A", 16)
    rules = Counter(s["rule"] for s in r16["steals"])
    kd = sum(1 for s in r16["steals"] if s["victim_key_down"])
    ivs, total = vpo.over_intervals(r["voices"], 16)
    print("[%s] unlimited-pool peak demand %d at t=%.3f s; alive at that instant %d"
          % (label, r["max"], t, len(live)))
    print("[%s]   envelope bins at the peak: %s" % (label, dict(sorted(bins.items()))))
    print("[%s]   key_down %d / released %d ; top (note,key_down): %s"
          % (label, sum(1 for v in live if v.key_down), sum(1 for v in live if not v.key_down),
             notes.most_common(6)))
    print("[%s] 16-voice pool: %d steals (rules %s), %d of them hit a key-held voice"
          % (label, len(r16["steals"]), dict(sorted(rules.items())), kd))
    print("[%s]   unlimited-pool demand > 16: %d intervals, %.1f s in total"
          % (label, len(ivs), total))

    dumps = sys.argv[2]
    consts = vpo.read_code_constants()
    sp = (REPO / "scores/examples/moonlight_sonata_complete.score.json").resolve()
    events, sr, _ = vpo.load_events(sp, dumps, consts)

def db(x):
    return 10 * math.log10(x) if x > 0 else float("-inf")


def last_off(case):
    lo = 0
    for line in open(EVD / (case + ".txt"), encoding="utf-8"):
        if line.startswith("ev ") and " off " in line:
            lo = max(lo, int(line.split()[1]))
    return lo


def kv_summary(p):
    out = {}
    for line in open(p, encoding="utf-8"):
        out[line.split()[0]] = line.strip()
    return out


def pair(case):
    rows = list(csv.DictReader(open(RUNS / "pair" / (case + "_blocks.csv"), encoding="utf-8")))
    sr, bs = 48000, 512
    n = len(rows) * bs
    s0 = sum(float(r["base_sumsq"]) for r in rows)
    s1 = sum(float(r["fix_sumsq"]) for r in rows)
    sd = sum(float(r["diff_sumsq"]) for r in rows)
    p0 = max(float(r["base_peak"]) for r in rows)
    p1 = max(float(r["fix_peak"]) for r in rows)
    mx = max(float(r["diff_maxabs"]) for r in rows)
    ndiff = sum(1 for r in rows if float(r["diff_maxabs"]) > 0)
    first = next((int(r["block"]) for r in rows if float(r["diff_maxabs"]) > 0), None)
    lo_blk = last_off(case) // bs
    t0 = sum(float(r["base_sumsq"]) for r in rows if int(r["block"]) > lo_blk)
    t1 = sum(float(r["fix_sumsq"]) for r in rows if int(r["block"]) > lo_blk)
    summ = (RUNS / "pair" / (case + "_summary.txt")).read_text(encoding="utf-8").strip()
    return {
        "audio_s": n / sr,
        "rms_dbfs_base": db(s0 / (2 * n)), "rms_dbfs_fix": db(s1 / (2 * n)),
        "peak_dbfs_base": 20 * math.log10(p0) if p0 > 0 else float("-inf"),
        "peak_dbfs_fix": 20 * math.log10(p1) if p1 > 0 else float("-inf"),
        "diff_energy_re_base_db": db(sd / s0) if s0 > 0 else None,
        "diff_maxabs": mx, "diff_maxabs_dbfs": 20 * math.log10(mx) if mx > 0 else float("-inf"),
        "blocks_differing": ndiff, "blocks_total": len(rows),
        "first_diff_t_s": None if first is None else first * bs / sr,
        "tail_from_s": (lo_blk + 1) * bs / sr,
        "tail_energy_db_base": db(t0), "tail_energy_db_fix": db(t1),
        "harness_summary": summ,
    }

            "seconds_16plus": full * 512 / 48000.0}


def fmt(x, nd=2):
    if x is None:
        return "-"
    if isinstance(x, float):
        if math.isinf(x):
            return "-inf"
        return ("%%.%df" % nd) % x
    return str(x)

    near_any, near_loud = set(), set()
    for s in st:
        b = int(s["sample"]) // 512
        rng = range(b - 1, b + 20)
        near_any.update(rng)
        if float(s["victim_env_db"]) > -40:
            near_loud.update(rng)
    tot = sum(float(r["diff_sumsq"]) for r in rows)
    sh = lambda S: sum(float(r["diff_sumsq"]) for r in rows if int(r["block"]) in S) / tot
    t.append("\n## moonlight_complete_A: where does the base-vs-fix difference sit?\n")
    t.append("share of total |base-fix| energy inside the ~0.2 s after a base steal (any victim): %.3f "
             "(%d of %d blocks)\n" % (sh(near_any), len(near_any), len(rows)))
    t.append("share inside the ~0.2 s after a base steal whose victim was still > -40 dB: %.3f "
             "(%d of %d blocks)\n" % (sh(near_loud), len(near_loud), len(rows)))
    top = sorted(rows, key=lambda r: -float(r["diff_maxabs"]))[:10]
    sb = {int(s["sample"]) // 512: s for s in st}
    t.append("10 blocks with the largest max|base-fix| (and the base steal in that block, if any):\n")
    for r in top:
        b = int(r["block"])
        s = sb.get(b)
        t.append("  t=%8.3f s  max|diff| %.4f (%.1f dBFS)  base steal: %s\n" % (
            b * 512 / 48000, float(r["diff_maxabs"]), 20 * math.log10(float(r["diff_maxabs"])),
            "none" if s is None else "incoming %s steals same-note %s still at %.1f dB (released)"
            % (s["incoming_note"], s["victim_note"], float(s["victim_env_db"]))))
    mx = [float(r["diff_maxabs"]) for r in rows]
    t.append("blocks whose max|diff| exceeds: " + ", ".join(
        "%d dBFS: %d" % (th, sum(1 for m in mx if m > 0 and 20 * math.log10(m) > th))
        for th in (-20, -40, -60, -80, -100, -120)) + " (of %d)\n" % len(rows))
    t.append("\n## CPU: moonlight_complete_A through both VST3s, 874.6 s of audio, wall-clock processBlock sum\n"
             "## (single process, lock-step; runs 1-3 in order base->fix, 'swapped' runs load fix first)\n")
    for p in sorted(glob.glob(str(C2 / "runs/timing/*_summary.txt"))):
        s = rd(p)
        m = re.search(r"process_s base ([\d.]+) fix ([\d.]+)", s)
        name = os.path.basename(p).replace("_summary.txt", "")
        a, b = float(m.group(1)), float(m.group(2))
        if "swapped" in name:
            a, b = b, a     # first instance was the FIX bundle
        t.append("%-16s base %.3f s (%.3f%% of real time)  fix %.3f s (%.3f%%)\n"
                 % (name, a, 100 * a / 874.581, b, 100 * b / 874.581))
    hs = sorted({sha(p) for p in glob.glob(str(C2 / "runs/timing/moon_[0-9]_blocks.csv"))})
    t.append("repeat runs' per-block csv identical across the 3 repeats: %s\n" % (len(hs) == 1))
    for c in cases:
        t.append("\n### harness log (parameter automation as read back from the plug-in): %s\n" % c)
        log = rd(C2 / ("runs/pair/%s_log.txt" % c)).splitlines()
        t.append("\n".join(log[:24]) + ("\n  ... (%d more lines)\n" % (len(log) - 24) if len(log) > 24 else "\n"))
    write("wf1002_C2_plugin_before_after.txt", "".join(t))

    # 4. zombie recount ----------------------------------------------------
    t = [HDR, "\n## A. WF0925-V1 fm_zombie_check.py as shipped (HEAD rule), dumps from build-base CLI\n",
         rd(C2 / "fm_zombie_check_original.txt"),
         "\n## B. same method, base rule vs C2 fix rule (c2_fm_zombie_fix_model.py; only\n"
         "##    voice_pool_occupancy.stop_tail_off differs)\n", rd(C2 / "model_base_vs_fix.txt"),
         "\n## C. measured on the real FMPianoVoice code (mirror harness, 1024-voice = unlimited pool,\n"
         "##    peak of simultaneously active voices, envelope bins at that instant)\n"]
    for L in ("base", "fix"):
        t.append("[%s] " % L + rd(C2 / ("runs/mirror_%s_1024/moonlight_complete_A_summary.txt" % L)).replace("\n", " | ") + "\n")
    for L in ("base", "fix"):
        t.append("[%s 16-voice] " % L + rd(C2 / ("runs/mirror_%s_16/moonlight_complete_A_summary.txt" % L)).replace("\n", " | ") + "\n")
    write("wf1002_C2_zombie_recount.txt", "".join(t))

    # 5. method sources ----------------------------------------------------
    t = [HDR, "\n## method sources (they live only in <C2>; copied here verbatim for review)\n"]
    for rel in ("tree/tests/c2_fm_harness.cpp", "method/c2_make_events.py",
                "method/c2_fm_zombie_fix_model.py", "method/c2_analyze.py",
                "method/c2_write_evidence.py"):
        t.append("\n" + "=" * 78 + "\n== <C2>/%s  sha256 %s\n" % (rel, sha(C2 / rel)) + "=" * 78 + "\n")
        t.append(rd(C2 / rel))
    tail = rd(C2 / "tree/CMakeLists.txt")
    i = tail.find("# --- WF1002-C2")
    t.append("\n" + "=" * 78 + "\n== appended to <C2>/tree*/CMakeLists.txt\n" + "=" * 78 + "\n" + tail[i:])
    t.append("\n## event files (<C2>/events) sha256\n")
    for p in sorted(glob.glob(str(C2 / "events/*"))):
        t.append("%s  %s\n" % (sha(p), os.path.basename(p)))
    t.append(rd(C2 / "events/events_meta.json"))
    write("wf1002_C2_method_sources.txt", "".join(t))
