import collections, sys
def load(p):
    out = []
    for l in open(p, encoding='utf-8', errors='replace'):
        if l.startswith('['):
            l = l.rstrip('\n')
            l = l.replace('hostprobe_baseline', 'HP').replace('K1' + chr(92) + 'hostprobe' + chr(92), 'K1' + chr(92) + 'HP' + chr(92))
            out.append(l)
    return out
b = load(sys.argv[1]); n = load(sys.argv[2])
cb = collections.Counter(b); cn = collections.Counter(n)
print('baseline [..] lines', len(b), '/ new [..] lines', len(n))
missing = cb - cn
print('baseline lines missing or changed in the new run:', sum(missing.values()))
for l in missing: print('   -', l)
added = cn - cb
print('lines only in the new run:', sum(added.values()))
for l in added: print('   +', l)
